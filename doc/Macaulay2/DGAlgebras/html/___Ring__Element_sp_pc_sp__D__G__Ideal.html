RingElement % DGIdeal -- Reduce a ring element modulo a DG ideal

  • Operator: %
  • Usage:
    r = f % I
  • Inputs:
    • f, a ring element, A RingElement of I.dgAlgebra.natural, or an integer (coerced into I.dgAlgebra.natural before reduction).
    • I, an instance of the type DGIdeal,
  • Outputs:
    • r, a ring element, The normal form of f modulo I.natural.

Description

i1 : R = ZZ/101[x, y]

o1 = R

o1 : PolynomialRing
 
i2 : A = koszulComplexDGA R

o2 = {Ring => R                          }
      Underlying algebra => R[T   ..T   ]
                               1,1   1,2
      Differential => {x, y}

o2 : DGAlgebra
 
i3 : Anat = A.natural

o3 = Anat

o3 : PolynomialRing, 2 skew commutative variable(s)
 
i4 : I = dgIdeal(A, {x_Anat})

o4 = DGIdeal of Anat
     generators => | x |

o4 : DGIdeal
 
i5 : (x_Anat^2 % I) == 0

o5 = true
 
i6 : (y_Anat^2 % I) == y_Anat^2

o6 = true
